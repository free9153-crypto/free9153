async function joinRoom(){ try{ const code=document.getElementById("roomInput") .value.trim().toUpperCase(); if(!/^[A-Z0-9]{6}$/.test(code)){ alert("방 코드 6자리를 입력해 주세요."); return; } const ref=db.ref("rooms/"+code); const result=await ref.transaction(room=>{ if(!room || room.status!=="waiting")return; room.players=room.players||{}; // 실제 참가자 목록을 기준으로 인원 계산 const ids=Object.keys(room.players); // 이미 참가한 사람은 다시 인원으로 세지 않음 if(ids.includes(playerId)){ return room; } // 실제 참가자 5명일 때만 차단 if(ids.length>=5)return; room.players[playerId]={ name:playerName(), joinedAt:Date.now() }; room.playerOrder=ids.concat(playerId); return room; }); if(!result.committed){ const latest=(await ref.get()).val(); if(!latest){ alert("방이 존재하지 않아요."); }else if(latest.status!=="waiting"){ alert("이미 시작한 방이에요."); }else{ alert("현재 참가자 수: "+Object.keys(latest.players||{}).length+"명"); } return; } roomCode=code; showGame(); listenRoom(); }catch(e){ console.error(e); alert("입장 오류: "+e.message); } }
